from scipy.spatial import KDTree
from scipy.stats import mode

# define the KnnRegression
class KnnRegression:

    def __init__(self, k):
        self.k = k

    def fit(self, x, y):
        self.y_train_ = y
        # use the KDTree to save x
        self.x_train_kdtree_ = KDTree(x)
        return self

    def predict(self, x):
        # use the KDTree to find the neighbours, neighbours is a row index of train set
        _, neighbours = self.x_train_kdtree_.query(x, k=self.k)
        neighbours = neighbours.reshape(len(x), self.k) # (n, k)
        #  get the mean as the predicting result
        neighbour_labels = self.y_train_[neighbours]
        m= np.mean(neighbour_labels, axis=1, keepdims=False)
        return m

def train_test_split(x, y, train_size=0.6, random_state=None):
    RNG = np.random.default_rng(random_state)
    N = len(x)
    N_train = round(N*train_size)
    idx_train = RNG.choice(N, N_train, replace=False)
    idx_test = np.setdiff1d(np.arange(N), idx_train)
    RNG.shuffle(idx_test)
    return x[idx_train], x[idx_test], y[idx_train], y[idx_test]

# define the L-Fold Cross Validation (CV) class
class LFold:

    def __init__(self, n_splits):
        # init the Fold number
        self.n_splits=n_splits

    def get_n_splits(self,x= None, y= None ,group= None):
        return self.n_splits

    def split(self, x, y= None, group= None):
        result = []
        if y is None:
            flods =np.array_split(x,self.n_splits)

            for i in range(self.n_splits):
                train = flods.copy()
                test = [flods[i]]
                del train[i]
                result.append((np.concatenate(train), np.concatenate(test)))

        else:
            # Combining x and y and get the folds
            x_plus_Y = np.hstack((x, y.reshape(y.shape[0], 1)))
            folds = np.array_split(x_plus_Y, self.n_splits)
            
            # save the x_train, x_test, y_train and y_test of Each fold, put in the list [()] 
            for i in range(self.n_splits):
                train = folds.copy()
                test  = [folds[i]]
                del train[i]
                result.append((
                        np.concatenate(train)[:,:-1],  # X_train：All columns except the last
                        np.concatenate(test)[:,:-1],  # X_test
                        np.concatenate(train)[:,-1:],  # y_train：Take only the last column 
                        np.concatenate(test)[:,-1:]   # y_test
                    ))
        return result

# calculate the mse of test and train , and save the result
def df_stores(L, max_k):
   
    # Cross Validation
    results = []  # [KValue, LFoldID, MSE_train, MSE_test]
    kf = LFold(n_splits=L)   

    # get the different k to run on the different fold
    for k in range(1, max_k+1):
        
        for fold_id, (x_train, x_test, y_train, y_test) in enumerate(kf.split(diabetes.data, diabetes.target)):
            
            mse_train, mse_test = runKnn(k, x_train, x_test, y_train, y_test)

            # save the result
            results.append([k, fold_id, mse_train, mse_test])

    # turn to the numply array
    results = np.array(results, dtype=float)
    return results

def compute_mean_std(results):
 
    K_values = np.unique(results[:, 0]).astype(int)
    
    summary_list = []
    for k in K_values:
        rows = results[results[:, 0] == k]
        mse_train = rows[:, 2]
        mse_test = rows[:, 3]
        summary_list.append([
            k,
            np.mean(mse_train), np.std(mse_train),
            np.mean(mse_test), np.std(mse_test)
        ])
    
    return np.array(summary_list)

def runKnncv(kValue, x_train, x_test, y_train, y_test):
    knn = KDTree(x_train)

    # predicting train set
    _, neighbours = knn.query(x_train, k=kValue)
    neighbours = neighbours.reshape(len(x_train), kValue)
    y_hat_train = np.mean(y_train[neighbours], axis=1)

    # predicting test set
    _, neighbours = knn.query(x_test, k=kValue)
    neighbours = neighbours.reshape(len(x_test), kValue)
    y_hat_test = np.mean(y_train[neighbours], axis=1)

    return MSE(y_train, y_hat_train), MSE(y_test, y_hat_test)

# calculate the mean k in the cross validation
def find_mean_k(results):
    # get the all fold ids
    folds = np.unique(results[:, 1]).astype(int) 
    best_ks = []
    
    for fold_id in folds:
        # put the same fold id rows together
        fold_rows = results[results[:, 1] == fold_id]
        mse_tests = fold_rows[:, 3]
        best_idx = np.argmin(mse_tests)
        best_k = int(fold_rows[best_idx, 0])
        best_ks.append(best_k)
    # return a mean of the best k in cv
    return int(np.ceil(np.mean(best_ks)))

Best k (outer test set): 19 On a completely independent test set, the best model performance was achieved when k=19.

The difference between the two is only 1, indicating a high degree of consistency between the cross-validation selection and the final test set results, validating the reliability of cross-validation.

Factors Affecting Cross-Validation's Success in Approximating the Optimal Model¶

Sample Size

The larger the sample size, the more stable the cross-validation results and the more they represent the true data distribution. With a small sample size, the cross-validation variance will be greater.

Cross-Validation Folds

Too few folds: This results in an overly large validation set and a too small training set, leading to large estimation bias.

Too many folds: Although more training data is used, the validation set is too small, making the results susceptible to noise.

Data Distribution and Noise

If the data contains a lot of noise, a small k is more susceptible to noise, while a large k can smooth out the noise. Different levels of data noise can affect the selection of the optimal k.
