def plot_perceptron_boundary(model, label, linestyle):
    w = model.coef_[0]
    b = model.intercept_[0]
    x_points = np.linspace(X_test_scaled[:, 0].min() - 1, X_test_scaled[:, 0].max() + 1, 10)

    plt.plot(x_points, y_points, label=label, linestyle=linestyle)

# define the prediction functiion 
def predict(model, x):
    W1, b1, W2, b2 = model['W1'], model['b1'], model['W2'], model['b2']
    # Forward propagation
    z1 = x.dot(W1) + b1
    a1 = np.tanh(z1)
    z2 = a1.dot(W2) + b2
    # Prevent out-of-range values
    z2_stable = z2 - np.max(z2, axis=1, keepdims=True)
    #  get the softmax probability
    exp_scores = np.exp(z2_stable)   
    probs = exp_scores / np.sum(exp_scores, axis=1, keepdims=True)
    # Return the cluster with the highest probability
    return np.argmax(probs, axis=1)

# define the Neural Network Model
def NeuralNetworkModel(nn_input_dim, nn_hdim, X_train, y_train, X_test, y_test, num_examples, eta, epoch_max=40000, print_loss=False):
    # init the parameters
    np.random.seed(0)
    W1 = np.random.randn(nn_input_dim, nn_hdim) / np.sqrt(nn_input_dim)
    b1 = np.zeros((1, nn_hdim))
    W2 = np.random.randn(nn_hdim, nn_output_dim) / np.sqrt(nn_hdim)
    b2 = np.zeros((1, nn_output_dim))
    
    model = {}
    test_error_list = []
    # running the epoch
    for i in np.arange(0, epoch_max):
        
        # Forward propagation
        z1 = X_train.dot(W1) + b1
        a1 = np.tanh(z1)
        z2 = a1.dot(W2) + b2
        # Subtract the maximum value from each row to ensure numerical stability
        z2_stable = z2 - np.max(z2, axis=1, keepdims=True)
        #  get the softmax probability
        exp_scores = np.exp(z2_stable)
        probs = exp_scores / np.sum(exp_scores, axis=1, keepdims=True)
        
        # Backpropagation and Calculate the gradient of each parameter
        L3 = probs
        # error of output
        L3[range(num_examples), y_train] -= 1
        
        dW2 = (a1.T).dot(L3)
        db2 = np.sum(L3, axis=0, keepdims=True)
        
        det2 = L3.dot(W2.T) * (1 - np.power(a1, 2))
        dW1 = np.dot(X_train.T, det2)
        db1 = np.sum(det2, axis=0)
        
        dW2 += L2_lambda * W2
        dW1 += L2_lambda * W1
        
        # Gradient descent parameter update
        W1 += -eta * dW1
        b1 += -eta * db1
        W2 += -eta * dW2
        b2 += -eta * db2
        
        model = { 'W1': W1, 'b1': b1, 'W2': W2, 'b2': b2}
        
        # return the test error
        if i == epoch_max - 1:
            final_test_error = zero_one_loss(predict(model, X_test).flatten(), y_test)

    return final_test_error, model

Q2-IV¶

Explain the reason(s) responsible for such difference between Perceptron and a 3-layer NeuralNetwork by comparing the plots you generated in Steps II and III.¶

As shown in the figure above, the decision boundary of the perceptron model can only linearly separate the data, while the three-layer neural network can nonlinearly separate the different classes. This is because the perceptron model assumes that its training data is linearly separable, and its decision boundary is also linear.

However, real data is nonlinearly separable and requires nonlinear transformations for effective classification. Furthermore, three-layer neural networks can perform nonlinear transformations through nonlinear activation functions and hidden layers, enabling them to learn nonlinear data. This is why three-layer neural networks are better at classifying nonlinear data.
