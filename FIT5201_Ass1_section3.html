from scipy.stats import multivariate_normal

# define the BayesianClassifier model
class BayesianClassifier:

    def __init__(self, shared_cov=True, cond_ind=True):
        # shared_cov: whether all classes share the same covariance matrix
        self.shared_cov = shared_cov
        # cond_ind: whether Feature is the conditional independence
        self.cond_ind = cond_ind

    # train the model
    def fit(self, x, y):
        # find all classes and return class_counts
        self.classes_, class_counts = np.unique(y, return_counts=True)
        
        # define the mean matrix and covariance matrix
        self.n_, self.p_ = x.shape
        self.k_ = len(self.classes_)
        self.cond_means_ = np.zeros(shape=(self.k_, self.p_))
        self.cond_covs_ = np.zeros(shape=(self.k_, self.p_, self.p_))
        #  prior probability matrix of each class
        self.class_priors_ = class_counts / len(y)

        # Loop over classes, compute mean per feature
        for c in range(self.k_):
            c_rows = y == c   # selects all rows of class c
            self.cond_means_[c, :] = x[c_rows].mean(axis=0)  # selects all rows in x that correspond to that class c.

            # Skip when the number of categories samples is too small to avoid singular matrices
            if np.sum(c_rows) < 2:
                self.cond_covs_[c, :, :] = np.eye(self.p_) * 1e-6
                continue
            
            # check Feature conditional independence
            if self.cond_ind:
                # Add the stable term to prevent variance to 0
                var = x[c_rows].var(axis=0)
                var[var == 0] = 1e-6                                                                                            
                #  The calculated variance fills the diagonal of this matrix     var(axis=0): calculate the variance
                np.fill_diagonal(self.cond_covs_[c, :, :], var)   
            else:
                # Add the stable term to prevent unstable covariance 
                cov = np.cov(x[c_rows].T, bias=True)  #cov(x[c_rows].T, bias=True) : calculate the covariance of each feature.
                cov += 1e-6 * np.eye(cov.shape[0]) 
                self.cond_covs_[c, :, :] = cov
                  
            
        # check and calculate shared covariance
        if self.shared_cov:
            shared_cov = (self.cond_covs_.T * self.class_priors_).T.sum(axis=0)
            self.cond_covs_[:] = shared_cov
        return self

    # predict probability
    def predict_proba(self, x):
        m, _ = x.shape
        cond_probs = np.zeros(shape=(m, self.k_))
        for c in range(self.k_):
            # calculate conditional_probs x\c
            cond_probs[:, c] = multivariate_normal.pdf(
                x, self.cond_means_[c], self.cond_covs_[c], allow_singular=True
            )
        # calculate marginal_probs  and avoid zero division
        marginal_probs = (cond_probs * self.class_priors_).sum(axis=1)+ 0.0001  
        # calculate posterior probability p(c|x)
        probs = ((cond_probs * self.class_priors_).T / marginal_probs).T   
        return probs
    
    def predict(self, x):
        return np.argmax(self.predict_proba(x), axis=1) # predict result is the max probability

    def decision_function(self, x):
        probs = self.predict_proba(x)
        # binary classification, return log odds
        # p(c=1|x) > p(c=0|x) or p(c=0|x) > p(c=1|x)
        if self.k_ == 2:
            return np.log(probs[:, 1] / probs[:, 0])
        else:
            res = np.zeros((len(x), self.k_))
            # more than two classes
            for c in range(self.k_):
                res[:, c] = np.log(probs[:, c] / (1 - probs[:, c]))
            return res
    # generate N data from class c Gaussian distribution.
    def generate(self, n, c, random_state=None):
        return multivariate_normal.rvs(
            self.cond_means_[c], self.cond_covs_[c], size=n, random_state=random_state)

import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.datasets import load_breast_cancer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import zero_one_loss

# define the array to save the result
#  [model_id, train_size, repeat_id, zero_one_train, zero_one_test]
results = []

# load the dataset
X, y = load_breast_cancer(return_X_y=True)

# Outer loop: different training set sizes
for n in range(5, 501, 5):
    # Inner loop: Repeat 10 times for each training set size
    for rep in range(10):
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, train_size=n, shuffle=True, random_state=None
        )

        # train four models
        naiveb_shared = BayesianClassifier(shared_cov=False, cond_ind=True).fit(X_train, y_train)
        bc_shared = BayesianClassifier(shared_cov=True, cond_ind=False).fit(X_train, y_train)
        bc = BayesianClassifier(shared_cov=False, cond_ind=False).fit(X_train, y_train)
        # check sampling of training sets with less than two classes
        if len(set(y_train)) < 2:
            # If the training set has only one class, skip Logistic regression
            models = [naiveb_shared, bc_shared, bc]
            model_names = ['NB(shared)', 'BC_shared', 'BC']
        else:
            lr = LogisticRegression(solver="lbfgs", max_iter=4800).fit(X_train, y_train)
            models = [naiveb_shared, bc_shared, bc, lr]
            model_names = ['NB(shared)', 'BC_shared', 'BC', 'LR']

        # record the result of each model
        for mid, model in enumerate(models):
            train_err = zero_one_loss(y_train, model.predict(X_train))
            test_err  = zero_one_loss(y_test,  model.predict(X_test))
            results.append([mid, n, rep, train_err, test_err])

# turn to the NumPy array
results = np.array(results, dtype=float)

Q5-IV¶

Q5-IV-a

As can be seen from the figure, as the training data increases, the training performance of each algorithm will gradually decrease, the test performance will gradually increase and eventually the performance will tend to be stable.

Q5-IV-b

As can be seen from the figure, when the training set is small, the performance of the LogisticRegression model test is better, and when the training set is large, the performance of the BayesianClassifier model with shared covariance is better.

Q5-IV-c

Because LogisticRegression is a simple model, while the Bayesian Classifier model with shared covariance is more complex, they must balancebias and variance to achieve better test set performance and prevent overfitting.

Therefore, the simpler LogisticRegression model can achieve good and stable performance with only a small training set. Similarly, the more complex Bayesian Classifier model requires a large training set to achieve good performance and prevent overfitting due to a small training set.
